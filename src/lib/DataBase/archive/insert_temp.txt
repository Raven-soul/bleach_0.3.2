insert into c_ticket_data(
    ticket_id,
    data_type,
    name,
    requirements,
    value,
    spoiler_id
)
select rt.ticket_id,
       rd.data_type,
       rd.name,
       rd.requirements,
       rd.value,
       rd.spoiler_id
  from c_class_ticket rt
       inner join cnt_class_data rd on rd.class_id = rt.id
